cmake_minimum_required(VERSION 3.20)
project(aether-security-center LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(aether-security-center main.cpp)
target_link_libraries(aether-security-center PRIVATE Qt6::Widgets)
target_compile_options(aether-security-center PRIVATE -fstack-protector-strong -D_FORTIFY_SOURCE=3)
target_link_options(aether-security-center PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS aether-security-center RUNTIME DESTINATION bin)
