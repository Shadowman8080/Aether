cmake_minimum_required(VERSION 3.25)
project(AetherGreeter VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg QuickWidgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIGHTDM REQUIRED IMPORTED_TARGET liblightdm-qt6-3)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
add_executable(aether-greeter main.cpp keyboard-state.cpp)
target_link_libraries(aether-greeter PRIVATE Qt6::Widgets Qt6::Svg Qt6::QuickWidgets PkgConfig::LIGHTDM PkgConfig::X11)
install(TARGETS aether-greeter RUNTIME DESTINATION bin)
install(FILES Keyboard.qml DESTINATION share/aether/greeter)
