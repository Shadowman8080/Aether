cmake_minimum_required(VERSION 3.25)
project(AetherVector VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
add_executable(vector main.cpp)
target_link_libraries(vector PRIVATE Qt6::Widgets KF6::Service KF6::KIOGui)
target_compile_options(vector PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
target_link_options(vector PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS vector RUNTIME DESTINATION bin)
option(AETHER_TEST_ICON_MOTION "Build the offscreen icon animation regression check" OFF)
if(AETHER_TEST_ICON_MOTION)
    add_executable(test-icon-motion test-icon-motion.cpp)
    target_link_libraries(test-icon-motion PRIVATE Qt6::Widgets)
endif()
