cmake_minimum_required(VERSION 3.21)
project(aether-assistant VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
qt_add_executable(aether-assistant main.cpp)
target_link_libraries(aether-assistant PRIVATE Qt6::Widgets)
install(TARGETS aether-assistant RUNTIME DESTINATION bin)
install(PROGRAMS aether-ai.py DESTINATION bin RENAME aether-ai)
install(FILES model.json DESTINATION share/aether-assistant)
install(FILES org.aether.Assistant.desktop DESTINATION share/applications)
install(FILES aether-assistant.svg DESTINATION share/icons/hicolor/scalable/apps)
