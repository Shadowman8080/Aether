cmake_minimum_required(VERSION 3.25)
project(AetherInstaller VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
add_executable(aether-installer main.cpp)
target_link_libraries(aether-installer PRIVATE Qt6::Widgets Qt6::Svg)
target_compile_options(aether-installer PRIVATE -Wall -Wextra -fstack-protector-strong)
target_link_options(aether-installer PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS aether-installer RUNTIME DESTINATION bin)
add_executable(aether-render-svg render-svg.cpp)
target_link_libraries(aether-render-svg PRIVATE Qt6::Gui Qt6::Svg)
