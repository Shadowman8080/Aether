cmake_minimum_required(VERSION 3.25)
project(AetherNimbrel VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(nimbrel main.cpp)
target_link_libraries(nimbrel PRIVATE Qt6::Widgets)
target_compile_options(nimbrel PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
target_link_options(nimbrel PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS nimbrel RUNTIME DESTINATION bin)
install(PROGRAMS client.py DESTINATION bin RENAME nimbrel-client)
install(FILES org.aether.Nimbrel.desktop DESTINATION share/applications)
install(FILES org.aether.Nimbrel.svg DESTINATION share/icons/hicolor/scalable/apps)

set(CMAKE_AUTOMOC ON)
find_package(KF6Runner REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(krunner_nimbrel MODULE runner.cpp)
target_link_libraries(krunner_nimbrel PRIVATE KF6::Runner KF6::CoreAddons Qt6::Gui)
install(TARGETS krunner_nimbrel LIBRARY DESTINATION plugins/kf6/krunner)
