cmake_minimum_required(VERSION 3.25)
project(AetherServerInstaller VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
add_executable(aether-server-installer main.cpp)
target_link_libraries(aether-server-installer PRIVATE Qt6::Widgets Qt6::Svg)
target_compile_options(aether-server-installer PRIVATE -Wall -Wextra -fstack-protector-strong)
target_link_options(aether-server-installer PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS aether-server-installer RUNTIME DESTINATION bin)
